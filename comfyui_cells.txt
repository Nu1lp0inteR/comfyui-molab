def dl(url, subdir, filename=None, headers=None):
    """下载到 models/<subdir>/，带断点续传。

    url     : 任意直链（HF / ModelScope / Civitai / GitHub Release 都行）
    subdir  : checkpoints / vae / clip / loras / controlnet / upscale_models
    filename: 想改名才填，否则自动取 URL 最后一段
    headers : 需要鉴权时传，如 {"Authorization": "Bearer hf_xxx"}
    """
    dest_dir = MODELS / subdir
    dest_dir.mkdir(parents=True, exist_ok=True)
    name = filename or urllib.parse.unquote(url.split("?")[0].rstrip("/").split("/")[-1])
    dest = dest_dir / name

    MIN = 1024 * 1024
    if dest.exists() and dest.stat().st_size > MIN:
        return f"已存在，跳过：`{subdir}/{name}` ({dest.stat().st_size / 2**30:.2f} GB)"

    cmd = ["curl", "-L", "-C", "-", "--retry", "3", "--retry-delay", "2", "-o", str(dest)]
    for _k, _v in (headers or {}).items():
        cmd += ["-H", f"{_k}: {_v}"]
    _r = subprocess.run(cmd + [url], capture_output=True, text=True)

    if not dest.exists():
        _req = urllib.request.Request(url, headers=headers or {})
        with urllib.request.urlopen(_req, timeout=120) as _res, open(dest, "wb") as _f:
            shutil.copyfileobj(_res, _f)

    if dest.stat().st_size < MIN:
        with open(dest, "rb") as _f:
            _head = _f.read(200).decode("utf-8", "ignore").strip()
        dest.unlink(missing_ok=True)
        return (
            f"❌ `{name}` 下载失败，返回的是文本而不是模型（已删除）：\n"
            f"```\n{_head[:150]}\n```\n"
            f"通常是链接写错（blob/main）、文件改名了、或需要登录——需要鉴权就加 headers 参数。"
        )
    return f"完成：`{subdir}/{name}` ({dest.stat().st_size / 2**30:.2f} GB)"

# ===== 第 9 格 =====
NODES = [
    "https://github.com/ltdrdata/ComfyUI-Manager",
    # "https://github.com/cubiq/ComfyUI_essentials",
    # "https://github.com/Fannovel16/comfyui_controlnet_aux",
    # "https://github.com/kijai/ComfyUI-KJNodes",
]

def install_node(url):
    name = url.rstrip("/").split("/")[-1]
    dest = COMFY_DIR / "custom_nodes" / name
    if not dest.exists():
        subprocess.run(["git", "clone", "--depth", "1", url, str(dest)], check=True)
    req = dest / "requirements.txt"
    if req.exists():
        subprocess.run(
            [sys.executable, "-m", "pip", "install", "-q", "-r", str(req)],
            check=True,
        )
    return f"`{name}` 已就绪"

def install_nodes():
    out = [install_node(u) for u in NODES]
    gpu = subprocess.run(
        [sys.executable, "-c", "import torch;print(torch.cuda.is_available())"],
        capture_output=True,
        text=True,
    ).stdout.strip()
    warn = ""
    if gpu != "True":
        warn = "\n\n⚠️ torch CUDA 变成不可用了——多半是某个节点的依赖把 cu128 换成了 CPU 版，重跑安装 ComfyUI 那一格即可恢复。"
    return mo.md("\n".join(out) + f"\n\ntorch CUDA 可用：`{gpu}`" + warn)

install_nodes()

def post(path, payload=None):
    req = urllib.request.Request(
        f"{COMFY_URL}{path}",
        data=None if payload is None else json.dumps(payload).encode(),
        headers={"Content-Type": "application/json"},
    )
    with urllib.request.urlopen(req, timeout=30) as r:
        return json.loads(r.read())

if not _ckpt.exists():

split-file 模型要在画布上自己搭：`UNETLoader`（选 diffusion_models）→
`CLIPLoader`（选 text_encoders）→ `VAELoader`（选 vae）→ `KSampler` → `VAEDecode` → `SaveImage`。

搭好之后在画布上用 `Save (API Format)` 导出 JSON，发我，我把这一格改成对应结构。
