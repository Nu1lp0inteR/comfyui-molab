def dl(url, subdir, filename=None, headers=None):
    """下载到 models/<subdir>/，带断点续传 + 坏文件识别。

    url     任意直链（HF / ModelScope / Civitai / GitHub Release）
    subdir  diffusion_models / text_encoders / vae / checkpoints / loras ...
    filename 想改名才填，否则取 URL 最后一段
    headers 需要鉴权时传，如 {"Authorization": "Bearer hf_xxx"}
    """
    _dir = MODELS / subdir
    _dir.mkdir(parents=True, exist_ok=True)
    _name = filename or urllib.parse.unquote(url.split("?")[0].rstrip("/").split("/")[-1])
    _dest = _dir / _name

    MIN = 1024 * 1024  # 小于 1MB 一律视为"没下成"
    if _dest.exists() and _dest.stat().st_size > MIN:
        return f"⏭ `{subdir}/{_name}` 已存在（{_dest.stat().st_size / 2**30:.2f} GB）"

    _cmd = ["curl", "-L", "-C", "-", "--retry", "3", "--retry-delay", "2", "-o", str(_dest)]
    for _k, _v in (headers or {}).items():
        _cmd += ["-H", f"{_k}: {_v}"]
    subprocess.run(_cmd + [url], capture_output=True, text=True)

    if not _dest.exists():
        _req = urllib.request.Request(url, headers=headers or {})
        with urllib.request.urlopen(_req, timeout=120) as _res, open(_dest, "wb") as _f:
            shutil.copyfileobj(_res, _f)

    if _dest.stat().st_size < MIN:
        with open(_dest, "rb") as _f:
            _head = _f.read(200).decode("utf-8", "ignore").strip()
        _dest.unlink(missing_ok=True)
        return f"❌ `{_name}` 返回的是文本不是模型（已删除）：`{_head[:120]}`"

    return f"✅ `{subdir}/{_name}`（{_dest.stat().st_size / 2**30:.2f} GB）"

            f"`{_p.name}` · {_p.stat().st_size / 1024:.0f} KB · 耗时 {time.time() - _t0:.0f} 秒\n\n"
            f"{_fw}×{_fh}（{_tag}）· 基础图另存为 `molab_base_*.png`\n\n"
            f"**种子 {_used_seed}** —— 想复现这一张，把 `SEED` 改成这个数字再跑。"
        ))
        _parts.append(mo.image(str(_p), width=_view_w))
        _parts.append(mo.download(
            _p.read_bytes(), filename=_p.name, label=f"⬇ 下载原图（{_fw}×{_fh}）"
        ))
        _out = mo.vstack(_parts)
except Exception as _e:
    # ComfyUI 的 400 响应体里，"node_errors" 才是真正的错误原因，
    # 外层 error.message 只有一句 "Prompt outputs failed validation"。
    _raw = ""
    try:
        _raw = _e.read().decode("utf-8", "ignore")
    except Exception:
        pass

    _lines = []
    try:
        _body = json.loads(_raw)
        for _nid, _ne in (_body.get("node_errors") or {}).items():
            _ct = _ne.get("class_type", "?")
            for _it in _ne.get("errors", []):
                _lines.append(
                    f"- **节点 {_nid}** (`{_ct}`)：{_it.get('message')}"
                    + (f"\n  - 细节：{_it.get('details')}" if _it.get("details") else "")
                )
    except Exception:
        pass

    if not _lines:
        _lines = ["（没能解析出 node_errors，原始响应见下方）"]

    _out = mo.md(
        "## 第 11 格 · 提交失败\n\n"
        + "\n".join(_lines)
        + "\n\n**常见原因对照**：\n"
        "- `Value not in list: sampler_name` → 把 `SAMPLER` 改成 `euler`\n"
        "- `Required input is missing` → 该节点有新的必填参数，把 `details` 里的名字告诉我\n"
        "- `Value not in list: unet_name / clip_name / vae_name` → 文件名和 `models/` 里的不一致\n"
        "- `Cannot connect to host` → ComfyUI 没启动，先跑第 10 格\n\n"
        f"```\n{_e}\n{_raw[:2000]}\n```"
    )

_out
