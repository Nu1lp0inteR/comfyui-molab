def dl(url, subdir, filename=None, headers=None):
    """下载到 models/<subdir>/，带断点续传。

    url     : 任意直链（HF / ModelScope / Civitai / GitHub Release 都行）
    subdir  : checkpoints / vae / clip / loras / controlnet / upscale_models
    filename: 想改名才填，否则自动取 URL 最后一段
    headers : 需要鉴权时传，如 {"Authorization": "Bearer hf_xxx"}
    """
    dest_dir = MODELS / subdir
    dest_dir.mkdir(parents=True, exist_ok=True)
    name = filename or urllib.parse.unquote(url.split("?")[0].rstrip("/").split("/")[-1])
    dest = dest_dir / name

    MIN = 1024 * 1024
    if dest.exists() and dest.stat().st_size > MIN:
        return f"已存在，跳过：`{subdir}/{name}` ({dest.stat().st_size / 2**30:.2f} GB)"

    cmd = ["curl", "-L", "-C", "-", "--retry", "3", "--retry-delay", "2", "-o", str(dest)]
    for _k, _v in (headers or {}).items():
        cmd += ["-H", f"{_k}: {_v}"]
    _r = subprocess.run(cmd + [url], capture_output=True, text=True)

    if not dest.exists():
        _req = urllib.request.Request(url, headers=headers or {})
        with urllib.request.urlopen(_req, timeout=120) as _res, open(dest, "wb") as _f:
            shutil.copyfileobj(_res, _f)

    if dest.stat().st_size < MIN:
        with open(dest, "rb") as _f:
            _head = _f.read(200).decode("utf-8", "ignore").strip()
        dest.unlink(missing_ok=True)
        return (
            f"❌ `{name}` 下载失败，返回的是文本而不是模型（已删除）：\n"
            f"```\n{_head[:150]}\n```\n"
            f"通常是链接写错（blob/main）、文件改名了、或需要登录——需要鉴权就加 headers 参数。"
        )
    return f"完成：`{subdir}/{name}` ({dest.stat().st_size / 2**30:.2f} GB)"

def _stats():
    try:
        with urllib.request.urlopen(f"{COMFY_URL}/system_stats", timeout=2) as r:
            return json.loads(r.read())
    except Exception:
        return None

LOG = COMFY_DIR.parent / "comfyui.log"
proc = None

if _stats() is None:
    proc = subprocess.Popen(
        [sys.executable, "main.py", "--listen", "127.0.0.1", "--port", "8188"],
        cwd=str(COMFY_DIR),
        stdout=open(LOG, "w"),
        stderr=subprocess.STDOUT,
    )
    for _ in range(120):
        if _stats() is not None:
            break
        time.sleep(2)

_s = _stats()
if _s:
    _dev = _s.get("devices", [{}])[0]
    mo.md(f"ComfyUI 已就绪：`{COMFY_URL}` · {_dev.get('name', '')} · VRAM {_dev.get('vram_total', 0) / 2**30:.0f} GB")
else:
    mo.md("启动失败，日志末尾：\n\n```\n" + LOG.read_text(errors="ignore")[-2000:] + "\n```")

def post(path, payload=None):
    req = urllib.request.Request(
        f"{COMFY_URL}{path}",
        data=None if payload is None else json.dumps(payload).encode(),
        headers={"Content-Type": "application/json"},
    )
    with urllib.request.urlopen(req, timeout=30) as r:
        return json.loads(r.read())

_pid = post("/prompt", {"prompt": wf})["prompt_id"]
_hist = {}
for _ in range(300):
    _hist = post(f"/history/{_pid}")
    if _pid in _hist:
        break
    time.sleep(2)

_imgs = _hist.get(_pid, {}).get("outputs", {}).get("9", {}).get("images", [])
if not _imgs:
    _have = [p.name for p in (MODELS / "checkpoints").iterdir()] if (MODELS / "checkpoints").exists() else []
    mo.md(f"没有产出。checkpoint 名字要跟 `models/checkpoints` 里完全一致，现有：{_have}")
else:
    _i = _imgs[0]
    _q = urllib.parse.urlencode({"filename": _i["filename"], "subfolder": _i.get("subfolder", ""), "type": _i.get("type", "output")})
    _p = MODELS.parent / _i["filename"]
    _p.write_bytes(urllib.request.urlopen(f"{COMFY_URL}/view?{_q}", timeout=60).read())
    mo.vstack([mo.md(f"`{_p.name}`"), mo.image(str(_p))])
